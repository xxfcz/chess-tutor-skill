<script>
const GLYPH={'P':'♙','N':'♘','B':'♗','R':'♖','Q':'♕','K':'♔',
             'p':'♟','n':'♞','b':'♝','r':'♜','q':'♛','k':'♚'};
let OPLIST=[], CUR=null;

async function post(b){
  const r=await fetch('/',{method:'POST',headers:{'Content-Type':'application/json'},
    body:JSON.stringify(b)});
  return r.json();
}
function parseFen(f){
  const g=[]; const rows=f.split(' ')[0].split('/');
  for(const row of rows){ const a=[];
    for(const ch of row){
      if(ch>='1'&&ch<='8'){ for(let i=0;i<+ch;i++) a.push(''); }
      else a.push(ch);
    } g.push(a);
  }
  return g;
}
function renderBoard(el,fen,sel,tgts){
  const g=parseFen(fen); el.innerHTML='';
  for(let r=0;r<8;r++) for(let c=0;c<8;c++){
    const sq=String.fromCharCode(97+c)+(8-r);
    const d=document.createElement('div');
    d.className='sq '+((r+c)%2?'d':'l');
    d.dataset.sq=sq;
    const pc=g[r][c];
    if(pc) d.textContent=GLYPH[pc]||'';
    if(sel===sq) d.classList.add('sel');
    if(tgts&&tgts.includes(sq)){ d.classList.add('tgt'); if(pc) d.classList.add('cap'); }
    el.appendChild(d);
  }
}

// ---------- 列表 ----------
async function loadList(){
  OPLIST=await post({action:'list'});
  const v=document.getElementById('view-list');
  let h='';
  for(const fam of ['e4','d4']){
    h+=`<div class="fam">${fam==='e4'?'e4 王兵体系':'d4 后兵体系'}</div><div class="opl">`;
    for(const o of OPLIST.filter(x=>x.family===fam)){
      h+=`<div class="op" data-id="${o.id}">
        <div>
          <div class="nm">${o.name} <span style="color:#64748b;font-size:11px">${o.side==='white'?'执白':'执黑'}</span></div>
          <div class="ln">${o.line}</div>
        </div>
        <span class="lv l${o.level}">${o.level===1?'入门':'进阶'}</span>
      </div>`;
    }
    h+='</div>';
  }
  v.innerHTML=h;
  v.querySelectorAll('.op').forEach(el=>{
    el.onclick=()=>{ CUR=OPLIST.find(x=>x.id===el.dataset.id); startDrill(CUR); };
  });
}

// ---------- 闯关 ----------
let D={sel:null,tgts:[],fen:null,plies:0,id:null};

async function startDrill(o){
  const r=await post({action:'drill_start', id:o.id});
  D={sel:null,tgts:[],fen:r.fen,plies:0,id:o.id,name:r.name,target:r.target,side:r.side};
  document.querySelector('[data-tab="drill"]').click();
  document.getElementById('view-drill').innerHTML=`
    <div class="card">
      <div class="bar"><span id="dname">${r.name}</span>
        <span id="dprog">0 / ${r.target}</span></div>
      <div class="prog"><i id="dbar" style="width:0%"></i></div>
      <div class="board" id="dboard"></div>
      <div id="dmsg"></div>
      <div style="display:flex;gap:8px;margin-top:10px">
        <button class="btn gh" id="dback">← 返回选择</button>
        <button class="btn" id="dretry">重来</button>
      </div>
    </div>
    <div class="card"><h2 style="font-size:15px;margin:0 0 8px">关键想法</h2>
      <ul class="ideas" id="dideas"></ul>
      <h2 style="font-size:15px;margin:14px 0 8px">易错点</h2>
      <ul class="ideas tr" id="dtraps"></ul>
    </div>`;
  drawD();
  const inf=await post({action:'ideas', id:o.id});
  document.getElementById('dideas').innerHTML=inf.ideas.map(x=>`<li>${x}</li>`).join('');
  document.getElementById('dtraps').innerHTML=inf.traps.map(x=>`<li>${x}</li>`).join('');
  document.getElementById('dback').onclick=()=>document.querySelector('[data-tab="list"]').click();
  document.getElementById('dretry').onclick=()=>startDrill({id:o.id});
}

function drawD(){
  const el=document.getElementById('dboard');
  renderBoard(el,D.fen,D.sel,D.tgts);
  el.querySelectorAll('.sq').forEach(sq=>{
    sq.onclick=()=>onDrillClick(sq.dataset.sq);
  });
  document.getElementById('dprog').textContent=`${D.plies} / ${D.target}`;
  document.getElementById('dbar').style.width=Math.min(100,D.plies/D.target*100)+'%';
}

async function onDrillClick(sq){
  if(!D.sel){
    D.sel=sq; D.tgts=[]; drawD(); return;
  }
  if(D.sel===sq){ D.sel=null; D.tgts=[]; drawD(); return; }
  const mv=D.sel+sq, from=D.sel;
  // 升变默认后
  let uci=mv; 
  const g=parseFen(D.fen);
  const rr=8-(+from[1]), cc=from.charCodeAt(0)-97;
  const pc=g[rr][cc];
  if(pc==='P' && sq[1]==='8') uci+='q';
  if(pc==='p' && sq[1]==='1') uci+='q';
  D.sel=null; D.tgts=[];
  const r=await post({action:'drill_move', id:D.id, fen:D.fen, move:uci, plies:D.plies});
  if(r.error){ msg('wr',r.error); drawD(); return; }
  D.fen=r.fen; D.plies=r.plies; drawD();
  let cls='nu', t='';
  if(r.verdict==='excellent'){ cls='ok'; t=`✓ ${r.san} 最佳！`; }
  else if(r.verdict==='good'){ cls='ok'; t=`○ ${r.san} 可行`; }
  else if(r.verdict==='offbook'){ cls='nu'; t=`◇ ${r.san} 不亏（${r.loss}cp），但不是本开局主变`; }
  else if(r.verdict==='inaccuracy'){ cls='nu'; t=`△ ${r.san} 略软（损失 ${r.loss}cp）`; }
  else { cls='wr'; t=`✗ ${r.san} 不好（损失 ${r.loss}cp）`; }
  if(r.expected_san && !r.in_book) t+=` · 谱内应走 <b>${r.expected_san}</b>`;
  if(r.best_san && r.verdict!=='excellent') t+=` · 引擎首选 <b>${r.best_san}</b>`;
  if(r.reply) t+=`<br><span style="color:#94a3b8">对手应：${r.reply}</span>`;
  if(r.reached_target) t+=`<br><b>★ 开局完成！</b>`;
  if(r.game_over) t+=`<br><b>对局结束</b>`;
  msg(cls,t);
}

function msg(cls,html){
  const el=document.getElementById('dmsg');
  if(el) el.innerHTML=`<div class="msg ${cls}">${html}</div>`;
}

// ---------- 默写 ----------
let Q={id:null,idx:0,ans:null,answered:false};

async function startQuiz(o){
  Q={id:o.id,idx:0,ans:null,answered:false};
  document.querySelector('[data-tab="quiz"]').click();
  document.getElementById('view-quiz').innerHTML=`
    <div class="card">
      <div class="bar"><span id="qname">${o.name}</span><span id="qprog"></span></div>
      <div class="prog"><i id="qbar" style="width:0%"></i></div>
      <div class="board" id="qboard" style="cursor:default"></div>
      <div style="font-size:13px;color:#94a3b8;margin-top:10px">下一步该走什么？</div>
      <div class="qopt" id="qopts"></div>
      <div id="qmsg"></div>
      <button class="btn gh" id="qback" style="margin-top:10px">← 返回选择</button>
    </div>`;
  document.getElementById('qback').onclick=()=>document.querySelector('[data-tab="list"]').click();
  nextQ();
}

async function nextQ(){
  const r=await post({action:'quiz_next', id:Q.id, idx:Q.idx});
  if(r.done){ 
    document.getElementById('qopts').innerHTML='';
    document.getElementById('qmsg').innerHTML='<div class="msg ok">★ 全部答完！换一条开局或点重练。</div>';
    return;
  }
  Q.ans=r.answer; Q.answered=false;
  renderBoard(document.getElementById('qboard'), r.fen, null, []);
  document.getElementById('qprog').textContent=`${Q.idx+1} / ${r.total}`;
  document.getElementById('qbar').style.width=((Q.idx)/r.total*100)+'%';
  document.getElementById('qmsg').innerHTML='';
  const box=document.getElementById('qopts'); box.innerHTML='';
  r.options.forEach(op=>{
    const d=document.createElement('div');
    d.className='qo'; d.textContent=op;
    d.onclick=()=>answerQ(d,op);
    box.appendChild(d);
  });
}

function answerQ(el,op){
  if(Q.answered) return;
  Q.answered=true;
  const right = op===Q.ans;
  el.classList.add(right?'right':'wrong');
  if(!right){
    document.querySelectorAll('#qopts .qo').forEach(x=>{
      if(x.textContent===Q.ans) x.classList.add('right');
    });
  }
  document.getElementById('qmsg').innerHTML = right
    ? '<div class="msg ok">✓ 正确！</div>'
    : `<div class="msg wr">✗ 应是 <b>${Q.ans}</b></div>`;
  setTimeout(()=>{ Q.idx++; nextQ(); }, right?900:1900);
}

document.querySelectorAll('.tab').forEach(t=>{
  t.onclick=()=>{
    document.querySelectorAll('.tab').forEach(x=>x.classList.remove('on'));
    t.classList.add('on');
    document.getElementById('view-list').classList.toggle('hide',t.dataset.tab!=='list');
    document.getElementById('view-drill').classList.toggle('hide',t.dataset.tab!=='drill');
    document.getElementById('view-quiz').classList.toggle('hide',t.dataset.tab!=='quiz');
  };
});

loadList();
