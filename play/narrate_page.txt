<script>
const feedEl=document.getElementById('feed');
function esc2(s){return String(s).replace(/[&<>]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;'}[c]));}
let lastLen=0;
async function pull(){
  try{
    const r=await fetch('/',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({action:'feed'})});
    const data=await r.json();
    const recs=Array.isArray(data)?data:(data.feed||[]);
    if(recs.length!==lastLen){
      lastLen=recs.length;
      render(recs);
    }
  }catch(e){/* 忽略瞬时错误 */}
}
function render(recs){
  feedEl.innerHTML='';
  for(const x of recs){ feedEl.appendChild(node(x)); }
  feedEl.scrollTop=feedEl.scrollHeight;
}
function node(x){
  const d=document.createElement('div');
  if(x.event==='move'){
    d.className='item move '+(x.color==='white'?'white':'black');
    d.innerHTML=`<span class="ply">${x.ply}.</span><span class="san">${x.san}</span>`;
  }else if(x.event==='coaching'){
    d.className='item coach';
    const labCls=x.label==='Blunder'?'Blunder':'Mistake';
    d.innerHTML=`<div><span class="lab ${labCls}">${x.label}${x.loss_cp!=null?' -'+x.loss_cp+'cp':''}</span>${x.phase||''}</div>`+
      (x.best?`<div>最佳着：${x.best}</div>`:'')+
      (x.reason?`<div>原因：${x.reason}</div>`:'')+
      (x.tailored?`<div style="color:var(--muted)">${x.tailored}</div>`:'');
  }else if(x.event==='coach_note'){
    d.className='item note';
    d.innerHTML=`<div class="who">🧑‍🏫 AI 教练 · ${x.ts||''}</div><div class="txt">${esc2(x.text||'')}</div>`;
  }else if(x.event==='narration'){
    d.className='item narr';
    d.innerHTML=`<div class="who">解说 · 第 ${x.ply||''} 手</div>${esc2(x.text||'')}`;
  }else if(x.event==='kpi'){
    const k=x.kpi||{};
    d.className='item kpi';
    d.textContent=`KPI — 最佳 ${k.Best||0} · 良好 ${k.Good||0} · 欠佳 ${k.Inaccuracy||0} · 失误 ${k.Mistake||0} · 大漏 ${k.Blunder||0} · 共 ${k.moves||0} 步`;
  }else if(x.event==='new_game'){
    d.className='item sep';d.textContent='—— 新局开始 ——';
  }else if(x.event==='game_over'){
    d.className='item sep';d.textContent='终局：'+(x.result||'');
  }else if(x.event==='sync'){
    d.className='item sep';d.textContent='已同步局面（'+x.moves+' 步）';
  }else if(x.event==='bridge_start'){
    d.className='item sep';d.textContent='观战桥已启动';
  }else{
    d.className='item sep';d.textContent=JSON.stringify(x);
  }
  return d;
}
pull();
setInterval(pull,2000);
</script>
